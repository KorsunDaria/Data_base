## 5. Предприятие

```mermaid
erDiagram
    СОТРУДНИК {
        int id_сотрудника
        string фио
        date дата_рождения
        int стаж_работы
    }

    ОБРАЗОВАНИЕ {
        int id_образования
        int id_сотрудника
        string уровень_образования
        string учебное_заведение
        int год_окончания
        string специальность
    }

    ДОЛЖНОСТЬ {
        int id_должности
        string название
    }

    ОТДЕЛ {
        int номер_отдела
        string название
    }

    НАЗНАЧЕНИЕ {
        int id_назначения
        int id_сотрудника
        int id_должности
        int номер_отдела
        decimal оклад
        string номер_приказа
        date дата_приказа
        date дата_начала
    }

    ОТПУСК {
        int id_отпуска
        int id_сотрудника
        date дата_начала
        date дата_окончания
    }

    БОЛЬНИЧНЫЙ_ЛИСТ {
        int id_листа
        int id_сотрудника
        string номер_листа
    }

    ПЕРИОД_НЕТРУДОСПОСОБНОСТИ {
        int id_периода
        int id_листа
        date дата_начала
        date дата_окончания
    }

    СОТРУДНИК ||--o{ ОБРАЗОВАНИЕ : "получил"
    СОТРУДНИК ||--|{ НАЗНАЧЕНИЕ : "назначается"
    ДОЛЖНОСТЬ ||--o{ НАЗНАЧЕНИЕ : "занимается"
    ОТДЕЛ ||--o{ НАЗНАЧЕНИЕ : "включает"
    СОТРУДНИК ||--o{ ОТПУСК : "берёт"
    СОТРУДНИК ||--o{ БОЛЬНИЧНЫЙ_ЛИСТ : "оформляет"
    БОЛЬНИЧНЫЙ_ЛИСТ ||--|{ ПЕРИОД_НЕТРУДОСПОСОБНОСТИ : "содержит"
```
